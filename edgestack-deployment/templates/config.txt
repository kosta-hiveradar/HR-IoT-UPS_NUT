# Disable Bluetooth (not needed, reduces attack surface)

# Disable onboard audio (not needed for edge datacenter use)
dtparam=audio=off

###############################################################################
# PERFORMANCE & STABILITY
###############################################################################

# Enable USB boot (if booting from USB storage instead of SD card)
# program_usb_boot_mode=1

# Increase GPU memory (default is too high for headless server)
# Minimum for headless: 16MB
# Note: Less GPU memory = more RAM for Docker containers

# Over-voltage for stability (Raspberry Pi 5 and 4)
# This provides better stability under load
# Values: 0-8 (0=default, 2=slight boost for stability, DO NOT exceed 6)

# CPU frequency governor (set to performance for consistent speed)
# This is handled by OS, but we can set boot frequency
# arm_freq=1800  # Uncomment for Raspberry Pi 4 (default is 1500-1800 depending on model)

# Improves response time for real-time monitoring

###############################################################################
# NOTES FOR HIVERADAR DEPLOYMENT
###############################################################################

# 1. WiFi and Bluetooth are DISABLED for security and stability
#    - Ethernet is the only supported network interface
#    - This prevents unauthorized wireless access
#    - Reduces power consumption and EM interference
#
# 2. GPU memory is MINIMIZED (16MB) to maximize RAM for Docker
#    - Headless server doesn't need GPU resources
#    - More RAM = more containers and better performance
#
# 3. USB is configured for MAXIMUM compatibility with UPS devices
#    - Higher current limit allows power-hungry UPS units
#    - USB quirks can be added per-device if needed
#
# 4. Boot is optimized for SPEED and reliability
#    - No unnecessary delays or splash screens
#    - UART enabled for emergency serial console access
#
# 5. Thermal management uses DEFAULT settings
#    - Ensure adequate cooling (heatsink + fan)
#    - Monitor temperatures via: vcgencmd measure_temp
#    - Critical for Raspberry Pi 5 under heavy load
#
# 6. HDMI is LEFT ENABLED by default
#    - Useful during initial setup and troubleshooting
#    - Can be disabled later to save minimal power
#
# 7. I2C is ENABLED for UPS devices that use it
#    - Some UPS units communicate via I2C
#    - No harm in leaving enabled even if not used
#
# 8. This configuration is TESTED on:
#    - Raspberry Pi 5 (8GB) - Primary deployment target
#    - Raspberry Pi 3B+ (1GB) - Legacy/low-cost deployments
#    - Raspberry Pi 4 (4GB/8GB) - Also compatible
#
# 9. DO NOT MODIFY these settings unless you know what you're doing:
#    - disable-wifi and disable-bt (required for security)
#    - gpu_mem=16 (critical for maximizing available RAM)
#    - max_usb_current=1 (needed for UPS reliability)

###############################################################################
# TROUBLESHOOTING
###############################################################################

# If you experience issues, try these diagnostic steps:
#
# 1. Boot issues:
#    - Connect HDMI monitor to see boot messages
#    - Check /boot/firmware/cmdline.txt for errors
#    - Verify SD card is not corrupted
#
# 2. USB UPS not detected:
#    - Verify max_usb_current=1 is set
#    - Try different USB port (use USB 3.0 ports if available)
#    - Check lsusb output: docker exec ups-nut lsusb
#    - Add USB quirk if specific UPS model has issues
#
# 3. Overheating/throttling:
#    - Check temperature: vcgencmd measure_temp
#    - Check throttling: vcgencmd get_throttled
#    - Install heatsink and/or fan
#    - Reduce over_voltage or set temp_limit lower
#
# 4. Network issues:
#    - Verify Ethernet cable is connected
#    - WiFi is disabled by design (dtoverlay=disable-wifi)
#    - Check /var/log/syslog for network errors
#
# 5. Slow performance:
#    - Check if throttling: vcgencmd get_throttled
#    - Verify adequate power supply (5V 3A minimum for Pi 4/5)
#    - Check available RAM: free -h
#    - Reduce number of running containers if needed
